IMAGE_EXTENSIONS = {".jpg", ".jpeg", ".png", ".bmp"}

def find_pairs(image_dir, xml_dir):
    images = {
        p.stem: p
        for p in image_dir.iterdir()
        if p.is_file() and p.suffix.lower() in IMAGE_EXTENSIONS
    }

    annotations = {
        p.stem: p
        for p in xml_dir.iterdir()
        if p.is_file() and p.suffix.lower() == ".xml"
    }

    common = sorted(set(images) & set(annotations))

    unmatched_images = sorted(set(images) - set(annotations))
    unmatched_xml = sorted(set(annotations) - set(images))

    pairs = [(images[name], annotations[name]) for name in common]

    return pairs, unmatched_images, unmatched_xml

train_pairs, train_no_xml, train_no_image = find_pairs(
    train_images_dir, train_xml_dir
)

test_pairs, test_no_xml, test_no_image = find_pairs(
    test_images_dir, test_xml_dir
)

if not train_pairs:
    raise RuntimeError("No matching training image/XML pairs found.")

if not test_pairs:
    raise RuntimeError("No matching test image/XML pairs found.")

def convert_annotation(xml_path, image_path, label_path):
    root = ET.parse(xml_path).getroot()

    with Image.open(image_path) as im:
        image_width, image_height = im.size

    xml_size = root.find("size")

    if xml_size is not None:
        xml_width = int(xml_size.findtext("width", image_width))
        xml_height = int(xml_size.findtext("height", image_height))
    else:
        xml_width, xml_height = image_width, image_height

    if xml_width <= 0 or xml_height <= 0:
        raise ValueError(f"Invalid XML image size: {xml_path}")

    scale_x = image_width / xml_width
    scale_y = image_height / xml_height

    labels = []

    for obj in root.findall("object"):
        class_name = (obj.findtext("name") or "").strip().lower()

        if class_name != "ship":
            continue

        box = obj.find("bndbox")
        if box is None:
            continue

        xmin = float(box.findtext("xmin")) * scale_x
        ymin = float(box.findtext("ymin")) * scale_y
        xmax = float(box.findtext("xmax")) * scale_x
        ymax = float(box.findtext("ymax")) * scale_y

        xmin = max(0.0, min(xmin, image_width))
        xmax = max(0.0, min(xmax, image_width))
        ymin = max(0.0, min(ymin, image_height))
        ymax = max(0.0, min(ymax, image_height))

        if xmax <= xmin or ymax <= ymin:
            continue

        x_center = ((xmin + xmax) / 2.0) / image_width
        y_center = ((ymin + ymax) / 2.0) / image_height
        width = (xmax - xmin) / image_width
        height = (ymax - ymin) / image_height

        labels.append(
            f"0 {x_center:.6f} {y_center:.6f} "
            f"{width:.6f} {height:.6f}"
        )

    label_path.write_text(
        "\n".join(labels) + ("\n" if labels else ""),
        encoding="utf-8"
    )

    return len(labels)

def get_pairs(image_dir, xml_dir):
    images = {
        p.stem: p for p in image_dir.iterdir()
        if p.is_file() and p.suffix.lower() in EXTENSIONS
    }
    xmls = {
        p.stem: p for p in xml_dir.glob("*.xml")
    }
    return [
        (images[k], xmls[k])
        for k in sorted(images.keys() & xmls.keys())
    ]

train_pairs = get_pairs(

def convert_xml(xml_path, image_path, label_path):
    root = ET.parse(xml_path).getroot()

    with Image.open(image_path) as im:
        width, height = im.size

    size = root.find("size")
    xml_w = int(size.findtext("width", width)) if size is not None else width
    xml_h = int(size.findtext("height", height)) if size is not None else height

    if xml_w <= 0 or xml_h <= 0:
        raise ValueError(f"Invalid dimensions in {xml_path}")

    sx, sy = width / xml_w, height / xml_h
    labels = []

    for obj in root.findall("object"):
        if (obj.findtext("name") or "").strip().lower() != "ship":
            continue

        box = obj.find("bndbox")
        if box is None:
            continue

        xmin = float(box.findtext("xmin")) * sx
        ymin = float(box.findtext("ymin")) * sy
        xmax = float(box.findtext("xmax")) * sx
        ymax = float(box.findtext("ymax")) * sy

        xmin = max(0, min(xmin, width))
        xmax = max(0, min(xmax, width))
        ymin = max(0, min(ymin, height))
        ymax = max(0, min(ymax, height))

        if xmax <= xmin or ymax <= ymin:
            continue

        xc = ((xmin + xmax) / 2) / width
        yc = ((ymin + ymax) / 2) / height
        bw = (xmax - xmin) / width
        bh = (ymax - ymin) / height

        labels.append(
            f"0 {xc:.6f} {yc:.6f} {bw:.6f} {bh:.6f}"
        )

    label_path.write_text(
        "\n".join(labels) + ("\n" if labels else ""),
        encoding="utf-8"
    )
    return len(labels)


for split, pairs in splits.items():
