    patch = land_mask[iy1:iy2, ix1:ix2]

import json
import pandas as pd
from pathlib import Path
from ultralytics import YOLO

PROJECT = Path("/content/drive/MyDrive/folder_name")

MODEL_PATH = (
    PROJECT / "SAR_Ship_Detection" /
    "ssdd_ship_detector" / "weights" / "best.pt"
)
DATA_YAML = PROJECT / "ssdd_yolo" / "data.yaml"

OUTPUT_DIR = (
    PROJECT / "SAR_Ship_Detection" / "validation_evaluation"
)
OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

model = YOLO(str(MODEL_PATH))

# Evaluate the validation split, not the test split.
metrics = model.val(
    data=str(DATA_YAML),
    split="val",
    imgsz=640,
    batch=8,
    conf=0.001,
    iou=0.6,
    plots=True,
    project=str(OUTPUT_DIR),
    name="ssdd_validation",
    exist_ok=True,
    verbose=True
)

precision = float(metrics.box.mp)
recall = float(metrics.box.mr)
map50 = float(metrics.box.map50)
map50_95 = float(metrics.box.map)

f1 = (
    2 * precision * recall / (precision + recall)
    if precision + recall > 0 else 0.0
)

rows = [
    {"dataset": "SSDD", "split": "validation",
     "metric": "Precision", "value": precision},
    {"dataset": "SSDD", "split": "validation",
     "metric": "Recall", "value": recall},
    {"dataset": "SSDD", "split": "validation",
     "metric": "F1-score", "value": f1},
    {"dataset": "SSDD", "split": "validation",
     "metric": "mAP@0.50", "value": map50},
    {"dataset": "SSDD", "split": "validation",
     "metric": "mAP@0.50:0.95", "value": map50_95}
]

df = pd.DataFrame(rows)
csv_path = OUTPUT_DIR / "validation_metrics.csv"
json_path = OUTPUT_DIR / "validation_metrics.json"

df.to_csv(csv_path, index=False)
with open(json_path, "w") as f:
    json.dump(rows, f, indent=2)

print("\nVALIDATION METRICS")
print(df.to_string(index=False))
print("\nCSV:", csv_path)
print("JSON:", json_path)
print("Evaluation plots:", metrics.save_dir)
