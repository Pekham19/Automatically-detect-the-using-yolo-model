import os

from google.colab import drive
from ultralytics import YOLO

drive.mount("/content/drive", force_remount=True)

MY_DRIVE = Path("/content/drive/MyDrive")
PROJECT_FOLDER = MY_DRIVE / "folder_name"

possible_paths = [
    PROJECT_FOLDER / "ssdd_yolo" / "data.yaml",
    PROJECT_FOLDER / "BBox_SSDD" / "data.yaml",
]

DATA_YAML = next(
    (p for p in possible_paths if p.is_file()),
    None
)

if DATA_YAML is None and PROJECT_FOLDER.is_dir():
    matches = list(PROJECT_FOLDER.glob("**/data.yaml"))
    DATA_YAML = next(
        (
            p for p in matches
            if (p.parent / "images" / "train").is_dir()
            and (p.parent / "labels" / "train").is_dir()
        ),
        None
    )

if DATA_YAML is None:
    raise FileNotFoundError(
        "Could not locate data.yaml with YOLO training folders. "
        "Check the Google Drive path and rerun the dataset conversion cell."
    )

DATASET_DIR = DATA_YAML.parent

for split in ["train", "val", "test"]:
    image_dir = DATASET_DIR / "images" / split
    label_dir = DATASET_DIR / "labels" / split

    if not image_dir.is_dir() or not label_dir.is_dir():
        raise FileNotFoundError(
            f"Missing image or label folder for split: {split}"
        )

    image_count = sum(
        1 for p in image_dir.iterdir()
        if p.suffix.lower() in {".jpg", ".jpeg", ".png", ".bmp"}
    )

    label_count = len(list(label_dir.glob("*.txt")))

    if image_count == 0 or image_count != label_count:
        raise ValueError(
            f"{split}: {image_count} images, {label_count} label files."
        )

print("Dataset verified:", DATASET_DIR)
print("Configuration:", DATA_YAML)

if not torch.cuda.is_available():
    raise RuntimeError(

from pathlib import Path
from ultralytics import YOLO
import json

# Paths
DATA_YAML = (
    "/content/drive/MyDrive/folder_name/"
    "ssdd_yolo/data.yaml"
)

BEST_MODEL = (
    "/content/drive/MyDrive/folder_name/"
    "SAR_Ship_Detection/ssdd_ship_detector/"
    "weights/best.pt"
)

OUTPUT_DIR = Path(
    "/content/drive/MyDrive/folder_name/"
    "SAR_Ship_Detection/test_evaluation"
)
OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

assert Path(DATA_YAML).is_file()
assert Path(BEST_MODEL).is_file()

model = YOLO(BEST_MODEL)

metrics = model.val(
    data=DATA_YAML,
    split="test",
    imgsz=640,
    batch=8,
    device=0,
    plots=True,
    project=str(OUTPUT_DIR),
    name="ssdd_test",
    exist_ok=True
)

# Extract metrics
results = {
    "precision": float(metrics.box.mp),
    "recall": float(metrics.box.mr),
    "mAP50": float(metrics.box.map50),
    "mAP50_95": float(metrics.box.map),
}

with open(OUTPUT_DIR / "test_metrics.json", "w") as f:
    json.dump(results, f, indent=4)

print("\nFINAL TEST METRICS")
for name, value in results.items():
    print(f"{name}: {value:.4f}")

print("\nResults saved to:", OUTPUT_DIR)


from pathlib import Path
